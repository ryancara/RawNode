cmake_minimum_required(VERSION 3.16)
project(JxlDngProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WITH_DNGSDK "Build the SDK bridge (requires the official SDK)" OFF)
set(LIBRAW_SOURCE_DIR "" CACHE PATH "Unmodified LibRaw 0.22.3 source")
set(DNG_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/dng_sdk" CACHE PATH "Adobe SDK source subset root (contains source/dng_host.h)")
set(LIBJXL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sources/libjxl" CACHE PATH "Pinned upstream libjxl 0.11.2 source")
if(NOT EXISTS "${LIBRAW_SOURCE_DIR}/libraw/libraw_version.h")
  message(FATAL_ERROR "Supply -DLIBRAW_SOURCE_DIR=/path/to/LibRaw-0.22.3")
endif()
find_package(ZLIB REQUIRED)
find_package(JPEG REQUIRED)
find_package(Threads REQUIRED)
# Equivalent source set to LibRaw's full library, excluding reduced-build stubs.
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS "${LIBRAW_SOURCE_DIR}/src/*.cpp")
list(FILTER raw_sources EXCLUDE REGEX "_ph\\.cpp$")
add_library(probe_raw STATIC ${raw_sources})
target_include_directories(probe_raw PUBLIC "${LIBRAW_SOURCE_DIR}")
target_compile_definitions(probe_raw PRIVATE USE_ZLIB USE_JPEG)
target_compile_definitions(probe_raw PUBLIC LIBRAW_NODLL)
target_link_libraries(probe_raw PUBLIC ZLIB::ZLIB JPEG::JPEG Threads::Threads)
if(WITH_DNGSDK)
  if(NOT EXISTS "${DNG_SDK_ROOT}/source/dng_host.h")
    message(FATAL_ERROR "SDK subset unavailable: supply DNG_SDK_ROOT containing source/dng_host.h")
  endif()
  if(NOT EXISTS "${LIBJXL_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Run prepare-libjxl.sh or supply the pinned LIBJXL_SOURCE_DIR")
  endif()
  # External upstream sources are pinned by prepare-libjxl.sh, never vendored here.
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  foreach(feature TOOLS DEVTOOLS JPEGLI JPEGLI_LIBJPEG DOXYGEN MANPAGES BENCHMARK
      EXAMPLES JNI SJPEG OPENEXR VIEWERS TCMALLOC PLUGINS)
    set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
  add_subdirectory("${LIBJXL_SOURCE_DIR}" bundled-jxl EXCLUDE_FROM_ALL)
  set(sdk_source "${DNG_SDK_ROOT}/source")
  file(GLOB sdk_sources CONFIGURE_DEPENDS "${sdk_source}/*.cpp")
  # This file is included directly by dng_lossless_jpeg.cpp, not compiled twice.
  list(FILTER sdk_sources EXCLUDE REGEX "/dng_lossless_jpeg_shared\\.cpp$")
  list(REMOVE_ITEM sdk_sources "${sdk_source}/dng_jxl.cpp")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare-sdk-compat.py"
    "${sdk_source}/dng_jxl.cpp" "${CMAKE_CURRENT_BINARY_DIR}/dng_jxl_no_xmp.cpp"
    RESULT_VARIABLE compat_status)
  if(NOT compat_status EQUAL 0)
    message(FATAL_ERROR "Could not prepare the SDK 2768 XMP-disabled compatibility source")
  endif()
  list(APPEND sdk_sources "${CMAKE_CURRENT_BINARY_DIR}/dng_jxl_no_xmp.cpp")
  add_library(probe_dng STATIC ${sdk_sources})
  target_include_directories(probe_dng PUBLIC "${sdk_source}")
  if(APPLE)
    set(sdk_platform qMacOS=1)
    find_library(sdk_corefoundation CoreFoundation)
    find_library(sdk_coreservices CoreServices)
    if(NOT sdk_corefoundation OR NOT sdk_coreservices)
      message(FATAL_ERROR "The macOS SDK requires CoreFoundation and CoreServices")
    endif()
    target_link_libraries(probe_dng PUBLIC ${sdk_corefoundation} ${sdk_coreservices})
  elseif(WIN32)
    set(sdk_platform qWinOS=1)
    target_link_libraries(probe_dng PUBLIC user32)
  else()
    set(sdk_platform qLinux=1)
  endif()
  target_compile_definitions(probe_dng PUBLIC ${sdk_platform} qDNGThreadSafe=1
    qDNGUseXMP=0 qDNGXMPFiles=0 qDNGXMPDocOps=0 qDNGUseLibJPEG=1 qDNGValidateTarget=0)
  target_link_libraries(probe_dng PUBLIC jxl jxl_threads JPEG::JPEG ZLIB::ZLIB Threads::Threads)
  target_compile_definitions(probe_raw PRIVATE USE_DNGSDK)
  target_link_libraries(probe_raw PUBLIC probe_dng)
endif()
add_executable(dng_probe src/probe.cpp)
target_compile_definitions(dng_probe PRIVATE POC_WITH_DNGSDK=$<BOOL:${WITH_DNGSDK}>)
target_link_libraries(dng_probe PRIVATE probe_raw)
enable_testing()
add_test(NAME memory_checks COMMAND dng_probe --selftest)
